/** Decorated abstract class with generic heritage */
@injectable()
export abstract class BaseService<T extends object, R = void> extends Repository<T> implements Identifiable, Loggable {
  private readonly id: string = "base";
  protected cache?: Map<string, T>;
  static instances: number = 0;
  abstract name: string;

  constructor(private options: Options, ...rest: string[]) {
    super();
  }

  get size(): number { return this.cache?.size ?? 0; }
  set size(v: number) { this.count = v; }

  async fetch(key: string): Promise<R | undefined> {
    const value = await this.cache?.get(key);
    return value as R | undefined;
  }

  protected abstract validate(input: T): input is T & { ok: true };
}
