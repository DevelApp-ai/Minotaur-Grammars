%% Erlang example corpus for Erlang.grammar (lexfile pin).
-module(my_app).
-export([start/0, add/2]).

-define(PI, 3.14159).

start() ->
    X = 42,
    Y = 3.14,
    Z = 16#FF,
    W = 1.0e-3,
    A = hello,
    B = 'quoted atom',
    C = "a string",
    D = $a,
    E = $\n,
    V = Variable,
    io:format("~p~n", [X]),
    add(X, Y),
    <<X:8, Y:16>> = <<1, 2>>,
    case X of
        0 -> zero;
        _ -> other
    end,
    if X > 0 -> pos; true -> neg end,
    receive M -> M after 1000 -> timeout end,
    fun(A, B) -> A + B end,
    [X || X <- [1, 2, 3]],
    lists:map(fun(X) -> X * 2 end, [1, 2]).

add(A, B) when is_integer(A) -> A + B;
add(A, B) -> A + B.
