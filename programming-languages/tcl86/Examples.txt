# fibonacci in Tcl
proc fib {n} {
    if {$n < 2} { return $n }
    return [expr {[fib [expr {$n - 1}]] + [fib [expr {$n - 2}]]}]
}

set names [list alice bob carol]
foreach name $names {
    puts "hello $name"
}

set total 0
for {set i 0} {$i < 10} {incr i} {
    set total [expr {$total + $i}]
}
puts "total=$total"
