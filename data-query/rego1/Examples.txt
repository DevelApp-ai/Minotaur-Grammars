# authz policy
package example.authz

import rego.v1

default allow := false

allow if {
    input.user == "admin"
}

allow if {
    some role in input.user.roles
    role == "editor"
    input.action == "write"
}

deny contains msg if {
    not allow
    msg := sprintf("denied: %v", [input])
}

users := {"anna": "admin", "bob": "editor"}
