# terraform-ish config
terraform {
  required_version = ">= 1.5"
}

resource "aws_instance" "web" {
  ami           = "ami-12345678"
  instance_type = "t3.micro"
  count         = 2
  tags = {
    Name = "web-${count.index}"
    Env  = "prod"
  }
}

variable "region" {
  default = "eu-west-1"
}

output "ip" {
  value = aws_instance.web[*].public_ip
}
