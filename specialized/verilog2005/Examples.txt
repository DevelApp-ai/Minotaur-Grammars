// half adder
module half_adder(input a, b, output sum, cout);
  assign sum = a ^ b;
  assign cout = a & b;
endmodule

`define WIDTH 8

module counter(input clk, input rst, output reg [`WIDTH-1:0] q);
  always @(posedge clk or posedge rst) begin
    if (rst) q <= 0;
    else q <= q + 1;
  end
endmodule

module tb;
  reg a, b;
  wire s, c;
  half_adder ha(.a(a), .b(b), .sum(s), .cout(c));
  initial begin
    a = 1'b1; b = 1'b0;
    $display("sum=%b cout=%b", s, c);
  end
endmodule
