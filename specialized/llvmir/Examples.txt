@count = global i32 0

define i32 @add(i32 %a, i32 %b) {
entry:
  %sum = add i32 %a, %b
  ret i32 %sum
}

define i32 @main() {
entry:
  %r = call i32 @add(i32 1, i32 2)
  %c = icmp eq i32 %r, 3
  br i1 %c, label %ok, label %bad
ok:
  ret i32 0
bad:
  ret i32 1
}
